from flask import Flask, request, render_template_string
import requests
from bs4 import BeautifulSoup

app = Flask(__name__)

# ---------------- HTML TEMPLATES ----------------

INDEX_HTML = """
<!DOCTYPE html>
<html>
<head>
    <title>Web Scraper</title>
    <style>
        body {
            font-family: Arial;
            background-color: #f2f2f2;
            text-align: center;
            margin-top: 100px;
        }
        input {
            width: 300px;
            padding: 10px;
        }
        button {
            padding: 10px 20px;
            background: green;
            color: white;
            border: none;
            cursor: pointer;
        }
    </style>
</head>
<body>

<h1>Web Scraper Working ✅</h1>

<form method="POST">
    <input type="text" name="content" placeholder="Enter product name" required>
    <br><br>
    <button type="submit">Search</button>
</form>

</body>
</html>
"""

RESULT_HTML = """
<!DOCTYPE html>
<html>
<head>
    <title>Reviews</title>
</head>
<body>

<h1>Amazon Reviews</h1>

{% for r in reviews %}
<div style="border:1px solid black; margin:10px; padding:10px;">
    <p><b>Name:</b> {{r.Name}}</p>
    <p><b>Rating:</b> {{r.Rating}}</p>
    <p><b>Title:</b> {{r.CommentHead}}</p>
    <p><b>Comment:</b> {{r.Comment}}</p>
</div>
{% endfor %}

<br>
<a href="/">Search Again</a>

</body>
</html>
"""

# ---------------- FLASK APP ----------------

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        search = request.form.get("content")

        url = f"https://www.amazon.in/s?k={search}"

        headers = {
            "User-Agent": "Mozilla/5.0",
            "Accept-Language": "en-US,en;q=0.9"
        }

        response = requests.get(url, headers=headers)
        soup = BeautifulSoup(response.content, "html.parser")

        links = soup.find_all("a", {"class": "a-link-normal s-no-outline"})

        if not links:
            return "No products found or Amazon blocked request"

        product_link = "https://www.amazon.in" + links[0]["href"]

        prod_page = requests.get(product_link, headers=headers)
        prod_soup = BeautifulSoup(prod_page.content, "html.parser")

        reviews = []

        review_blocks = prod_soup.find_all("div", {"data-hook": "review"})

        if not review_blocks:
            return "No reviews found"

        for r in review_blocks:
            reviews.append({
                "Name": r.find("span", {"class": "a-profile-name"}).text if r.find("span", {"class": "a-profile-name"}) else "N/A",
                "Rating": r.find("i", {"data-hook": "review-star-rating"}).text if r.find("i", {"data-hook": "review-star-rating"}) else "N/A",
                "CommentHead": r.find("a", {"data-hook": "review-title"}).text.strip() if r.find("a", {"data-hook": "review-title"}) else "N/A",
                "Comment": r.find("span", {"data-hook": "review-body"}).text.strip() if r.find("span", {"data-hook": "review-body"}) else "N/A"
            })

        return render_template_string(RESULT_HTML, reviews=reviews)

    return render_template_string(INDEX_HTML)


if __name__ == "__main__":
    app.run(debug=True)
