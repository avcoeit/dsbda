import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Load dataset
heart = pd.read_csv("heart disease.csv")

# Print column names
print(heart.columns)

# Create target column
if 'num' in heart.columns:
    heart['target'] = heart['num'].apply(lambda x: 0 if x == 0 else 1)

sns.set_style("whitegrid")

# Heatmap
plt.figure(figsize=(10,6))
sns.heatmap(heart.corr(numeric_only=True), annot=True, cmap='coolwarm')
plt.title("Correlation Heatmap")
plt.show()

# Countplot
if 'target' in heart.columns:
    sns.countplot(x='target', data=heart)
    plt.title("Heart Disease Count")
    plt.xlabel("0 = No Disease , 1 = Disease")
    plt.ylabel("Count")
    plt.show()

# Age Distribution
if 'age' in heart.columns:
    plt.figure(figsize=(8,5))
    sns.histplot(heart['age'], bins=20, kde=True)
    plt.title("Age Distribution")
    plt.xlabel("Age")
    plt.ylabel("Frequency")
    plt.show()

# Chest Pain vs Disease
if 'cp' in heart.columns and 'target' in heart.columns:
    sns.countplot(x='cp', hue='target', data=heart)
    plt.title("Chest Pain vs Heart Disease")
    plt.show()

# Pairplot
pair_columns = ['age', 'trestbps', 'chol']



sns.pairplot(heart, hue='target', vars=pair_columns)
plt.show()
