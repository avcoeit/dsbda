STEP 3 — Create mapper.py
------------------------------------------------------------
Command:
nano mapper.py

Code:
#!/usr/bin/env python3

import sys
from datetime import datetime

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue

    parts = line.split()
    if len(parts) < 4:
        continue

    username = parts[0]
    action = parts[1].upper()
    date_time = parts[2] + " " + parts[3]

    try:
        ts = datetime.strptime(date_time, "%Y-%m-%d %H:%M:%S").timestamp()
        print(f"{username}\t{action}\t{int(ts)}")
    except ValueError:
        continue

Save: CTRL + X → Y → ENTER


STEP 4 — Create reducer.py
------------------------------------------------------------
Command:
nano reducer.py

Code:
#!/usr/bin/env python3

import sys

login_times = {}
durations = {}

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue

    parts = line.split("\t")
    if len(parts) != 3:
        continue

    username = parts[0]
    action = parts[1]
    timestamp = int(parts[2])

    if action == "LOGIN":
        login_times[username] = timestamp

    elif action == "LOGOUT":
        if username in login_times:
            duration = timestamp - login_times[username]
            durations[username] = durations.get(username, 0) + duration
            del login_times[username]

print("\nLOGIN DURATION REPORT\n")

for user, secs in durations.items():
    hours = secs // 3600
    minutes = (secs % 3600) // 60
    print(f"{user} -> {hours}h {minutes}m")

max_user = max(durations, key=durations.get)

print(f"\nMAX LOGIN USER: {max_user}")


Save: CTRL + X → Y → ENTER
